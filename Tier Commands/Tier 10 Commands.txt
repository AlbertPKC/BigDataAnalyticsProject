# ---------------------------------------------------------------------
# ARTEFACT 1 - THE JOIN
# A real multi-table .join(), not a groupBy dressed up as one.
#
# A small synthetic lookup table classifies each of the 16 airports by
# hub size. It is joined TWICE against the main DataFrame: once on
# startingAirport and once on destinationAirport, so each itinerary is
# enriched with the hub size at both ends of the route.
#
# Because the lookup table is tiny, Spark broadcasts it to every
# executor rather than shuffling the 41M-row side. That is why this
# join costs far less than a shuffle join would.
# ---------------------------------------------------------------------
cat > tier10_step1_join.py << 'EOF'
import time
from pyspark.sql import SparkSession
from pyspark.sql import functions as F

spark = SparkSession.builder.appName("Tier10Step1Join").getOrCreate()
INPUT = "s3://aws-logs-518537549162-us-east-1/Final Project Files/itineraries_sample.csv"
df = spark.read.option("header", True).csv(INPUT)
df = df.withColumn("totalFare", F.col("totalFare").cast("double"))

# Synthetic airport hub-size lookup table
hub_data = [
    ("ATL", "major_hub"), ("ORD", "major_hub"), ("DFW", "major_hub"),
    ("LAX", "major_hub"), ("DEN", "major_hub"), ("JFK", "major_hub"),
    ("SFO", "mid_hub"),   ("EWR", "mid_hub"),   ("MIA", "mid_hub"),
    ("BOS", "mid_hub"),   ("CLT", "mid_hub"),   ("LGA", "mid_hub"),
    ("IAD", "small_hub"), ("PHL", "small_hub"), ("DTW", "small_hub"),
    ("OAK", "small_hub"),
]
hubs = spark.createDataFrame(hub_data, ["airport", "hub_size"])

t0 = time.time()

# Join 1: enrich with the ORIGIN airport hub size
enriched = df.join(
    hubs.withColumnRenamed("airport", "startingAirport")
        .withColumnRenamed("hub_size", "origin_hub_size"),
    on="startingAirport",
    how="left"
)

# Join 2: enrich with the DESTINATION airport hub size
enriched = enriched.join(
    hubs.withColumnRenamed("airport", "destinationAirport")
        .withColumnRenamed("hub_size", "dest_hub_size"),
    on="destinationAirport",
    how="left"
)

# Any airport not present in the lookup falls back to unclassified
enriched = (enriched
    .fillna({"origin_hub_size": "unclassified", "dest_hub_size": "unclassified"}))

result = (enriched
    .groupBy("origin_hub_size", "dest_hub_size")
    .agg(F.avg("totalFare").alias("avg_fare"),
         F.count("*").alias("n"))
    .orderBy("origin_hub_size", "dest_hub_size"))

result.show(50, truncate=False)
print(f"STEP1_JOIN elapsed: {time.time() - t0:.4f} sec")
spark.stop()
EOF

# ---------------------------------------------------------------------
# ARTEFACT 2 - THE WINDOW FUNCTION
# Ranks routes by average fare WITHIN each month and keeps the top 5.
#
# The key difference from a plain groupBy: a window function ranks rows
# inside a group WITHOUT collapsing the group. A groupBy would reduce
# each month to a single row; a window keeps every route row and adds a
# rank column alongside it.
# ---------------------------------------------------------------------
cat > tier10_step2_window.py << 'EOF'
import time
from pyspark.sql import SparkSession
from pyspark.sql import functions as F
from pyspark.sql.window import Window

spark = SparkSession.builder.appName("Tier10Step2Window").getOrCreate()
INPUT = "s3://aws-logs-518537549162-us-east-1/Final Project Files/itineraries_sample.csv"
df = spark.read.option("header", True).csv(INPUT)
df = df.withColumn("totalFare", F.col("totalFare").cast("double"))

t0 = time.time()

monthly = (df.withColumn("month", F.date_format(F.to_date("flightDate", "yyyy-MM-dd"), "yyyy-MM"))
             .groupBy("startingAirport", "destinationAirport", "month")
             .agg(F.avg("totalFare").alias("avg_fare")))

w = Window.partitionBy("month").orderBy(F.desc("avg_fare"))
ranked = monthly.withColumn("rank", F.rank().over(w))
ranked.filter(F.col("rank") <= 5).orderBy("month", "rank").show(50, truncate=False)

print(f"STEP2_WINDOW elapsed: {time.time() - t0:.4f} sec")
spark.stop()
EOF

# ---------------------------------------------------------------------
# ARTEFACT 3 - THE PHYSICAL PLAN
# This is the evidence artefact for the report. Save the output to
# /results/tier10_physical_plan.txt
#
# The plan has six steps. Read them top to bottom:
#   (1) Scan csv          - read only the 4 columns needed of the 27
#   (2) Project           - cast totalFare, derive the month string
#   (3) HashAggregate     - PARTIAL aggregate, computed on each node
#   (4) Exchange          - THE SHUFFLE. Data crosses the network here.
#   (5) HashAggregate     - FINAL aggregate, combining partial results
#   (6) AdaptiveSparkPlan - runtime re-optimisation wrapper
#
# Step 4 is the important one for the report. The Exchange is the
# mechanical reason a distributed aggregation costs more than a local
# one, and it explains why Q3 took longer than Q1 on every cluster
# tier. Steps 3 and 5 are why Spark beats MapReduce: the partial
# aggregate at step 3 means only per-node subtotals cross the network
# at step 4, not all 41 million individual fares.
# ---------------------------------------------------------------------
cat > tier10_step3_plan.py << 'EOF'
from pyspark.sql import SparkSession
from pyspark.sql import functions as F

spark = SparkSession.builder.appName("Tier10Step3Plan").getOrCreate()
INPUT = "s3://aws-logs-518537549162-us-east-1/Final Project Files/itineraries_sample.csv"
df = spark.read.option("header", True).csv(INPUT)

monthly = (df.withColumn("totalFare", F.col("totalFare").cast("double"))
             .withColumn("month", F.date_format(F.to_date("flightDate", "yyyy-MM-dd"), "yyyy-MM"))
             .groupBy("startingAirport", "destinationAirport", "month")
             .agg(F.avg("totalFare").alias("avg_fare")))

# explain() is NOT an action, so this costs almost nothing to run.
monthly.explain(mode="formatted")
spark.stop()
EOF
